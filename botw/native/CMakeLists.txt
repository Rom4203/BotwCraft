cmake_minimum_required(VERSION 3.16)
project(botwcraft_pose_controller C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(botwcraft_pose STATIC flight_controller.c)
target_include_directories(botwcraft_pose PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(NOT MSVC)
  target_link_libraries(botwcraft_pose PUBLIC m)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(botwcraft_pose_test tests/test_controller.c)
  target_link_libraries(botwcraft_pose_test PRIVATE botwcraft_pose)
  add_test(NAME pose_controller COMMAND botwcraft_pose_test)
endif()
